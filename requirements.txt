# Runtime dependencies — what the API + Streamlit need to SERVE predictions.
# Training-only deps live in `requirements-train.txt` (pulled in by
# requirements-dev.txt). This split keeps the production Docker image
# small (~300MB vs ~3GB) and means CI's pip-audit on this file only
# flags CVEs in code that actually ships to users.
#
# Dependabot proposes weekly patch and minor upgrades. Majors are ignored,
# and so are minor bumps for the packages with numpy in their tree.

# Core
pandas==2.2.3  # frozen: fitted into the shipped pickles
numpy==1.26.4  # frozen: fitted into the shipped pickles; also the numpy<2 ceiling
scikit-learn==1.8.0  # frozen: must equal the version the models were fitted under — a mismatch corrupts predictions silently
scipy==1.13.1  # frozen: 1.18.0 is the first release requiring numpy>=2, which makes this file unresolvable

# Gradient boosting candidates — compared on val in run_training.py, not shipped
xgboost==2.1.4  # frozen: fitted into the shipped pickles
lightgbm==4.6.0  # floor 4.6.0 fixes PYSEC-2024-231; a training-time candidate, not shipped

# NOTE: geopy, h3, pandera, and folium are deliberately absent —
# nothing in the tracked codebase imports any of them (haversine is
# hand-rolled in src/utils/geo.py; the dead H3/KMeans feature code that
# justified h3 never fed a model). Unused runtime deps are image weight
# plus CVE surface plus a false claim, all at once.

# API. fastapi's transitive pin controls starlette; newer fastapi brings

fastapi>=0.141.1,<1.0

starlette>=1.3.1

pydantic-settings==2.7.1
